  for kind in ['success-rates','clean-hold-distributions']:

  for r in new['rows']:
   c=current[r['trial_id']];m.require(r['profile_sha256']==c['profile_sha256'],'appendix profile changed');m.require(abs(r['actual_release_s']-c['actual_release_s'])<1e-8,'appendix release changed')
  rows=[r for r in old if r['route']==route and r['method']in METHODS]+new['rows'];stats={}
