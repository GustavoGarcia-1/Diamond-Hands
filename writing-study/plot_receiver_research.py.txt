"""Publication labels and explicitly post-hoc paired contrasts for the frozen450.

This consumes the independently rescored exported CSV; it does not alter trial
selection, criteria, or any recorded outcome. No training-seed inference is made.
"""
from pathlib import Path
import argparse
import csv
import hashlib
import json
import math
import numpy as np

METHODS=['current','earlier','exact_ball_diagnostic']
LABELS={
    'go':['Visual controller\n(0.30 s thumb delay)','Timing reference\n(0.50 s thumb delay)','Exact ball\nstate diagnostic'],
    'out':['Aligned visual\ncontroller','Prior pose / grip\nreference','Exact ball\nstate diagnostic'],
    'slant':['Lower palm\nvisual controller','Grip delay\nablation (+0.14 s)','Exact ball\nstate diagnostic'],
}
COLORS=['#202020','#999999','#376d7b']


def exact_mcnemar(wins,losses):
    n=wins+losses
    return 1. if n==0 else min(1.,2.*sum(math.comb(n,k)for k in range(min(wins,losses)+1))/2**n)


def main():
    parser=argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--input',type=Path,default=Path('site/dist/comparison/trials.csv'))
    parser.add_argument('--summary',type=Path,default=Path('site/dist/comparison/summary.json'))
    parser.add_argument('--output',type=Path,required=True)
    args=parser.parse_args();args.output.mkdir(parents=True,exist_ok=False)
    with args.input.open()as stream:rows=list(csv.DictReader(stream))
    assert len(rows)==450
    assert len({(r['route'],r['method'],r['trial_id'])for r in rows})==450
    summary=json.loads(args.summary.read_text());routes={r['route']:r for r in summary['routes']}
    exported=[];contrasts=[]
    for route in ['go','out','slant']:
        selected={m:sorted([r for r in rows if r['route']==route and r['method']==m],key=lambda x:int(x['trial_id']))for m in METHODS}
        for method in METHODS:
            records=selected[method]
            assert [int(r['trial_id'])for r in records]==list(range(50))
            observed=sum(r['primary_success']=='True'for r in records)
            known=routes[route]['methods'][method]
            assert observed==known['successes'] and known['total']==50
            exported.append(dict(route=route,method=method,successes=observed,total=50,rate_percent=2*observed,
                                 wilson95_percent=[100*x for x in known['wilson95']]))
        c=selected['current'];r=selected['earlier']
        wins=sum(x['primary_success']=='True'and y['primary_success']=='False'for x,y in zip(c,r))
        losses=sum(x['primary_success']=='False'and y['primary_success']=='True'for x,y in zip(c,r))
        both=sum(x['primary_success']=='True'and y['primary_success']=='True'for x,y in zip(c,r))
        known=routes[route]['paired']['current-minus-earlier']
        assert(wins,losses,both)==(known['candidate_only_success'],known['reference_only_success'],known['both_success'])
        contrasts.append(dict(route=route,current_only=wins,reference_only=losses,both_success=both,
                              both_failure=50-wins-losses-both,paired_difference_percentage_points=2*(wins-losses),
                              exact_mcnemar_two_sided_p=exact_mcnemar(wins,losses)))
    # One declared post-hoc family: current versus route-specific earlier for3routes.
    order=np.argsort([x['exact_mcnemar_two_sided_p']for x in contrasts]);running=0.
    for rank,index in enumerate(order):
        running=max(running,min(1.,(3-rank)*contrasts[index]['exact_mcnemar_two_sided_p']))
        contrasts[index]['holm_adjusted_p']=running
    result=dict(schema='receiver-research-presentation-v1',frozen450_outcomes_unchanged=True,
                source_csv_sha256=hashlib.sha256(args.input.read_bytes()).hexdigest(),
                source_summary_sha256=hashlib.sha256(args.summary.read_bytes()).hexdigest(),
                analysis_script_sha256=hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
                rates=exported,exploratory_paired_contrasts=contrasts,
                statistical_scope='Post-hoc exact conditional McNemar test on50paired seeded throws; Holm correction across3current-versus-earlier contrasts. Not a preregistered significance analysis, training-seed comparison, or hardware-reliability claim.',
                interval_scope='Marginal95%Wilson binomial intervals for the fixed per-route engineering sampler; not paired-effect intervals.',
                limitations=['Route-specific controller references differ; Out changes pose and grip together, preventing attribution to a single component.',
                             'Different physical models/horizons across routes; no pooled success rate.',
                             'The exact-ball diagnostic bypasses visibility/confidence and is not a learned-policy upper bound.',
                             'Training repeats are not represented by50throw perturbations.'])
    (args.output/'paired-statistics.json').write_text(json.dumps(result,indent=2)+'\n')
    import matplotlib
    matplotlib.use('Agg')
    import matplotlib.pyplot as plt
    plt.rcParams.update({'font.family':'sans-serif','font.sans-serif':['Arial','Helvetica','DejaVu Sans'],
                         'font.size':10,'axes.titlesize':12,'axes.spines.top':False,'axes.spines.right':False,
                         'svg.fonttype':'none','savefig.facecolor':'white'})
    for route in ['go','out','slant']:
        s=routes[route]
        fig,ax=plt.subplots(figsize=(5.2,4.3))
        rates=np.array([s['methods'][m]['rate']*100 for m in METHODS]);ci=np.array([s['methods'][m]['wilson95']for m in METHODS])*100
        bars=ax.bar(range(3),rates,color=COLORS,width=.55);bars[2].set_hatch('//')
        ax.errorbar(range(3),rates,yerr=np.vstack([rates-ci[:,0],ci[:,1]-rates]),fmt='none',color='#202020',capsize=4)
        ax.set(ylim=(0,100),ylabel='Retained receive success (%)',title=route.capitalize())
        ax.set_xticks(range(3),LABELS[route]);ax.tick_params(axis='x',labelsize=9);ax.grid(axis='y',alpha=.15);ax.set_axisbelow(True)
        for i,m in enumerate(METHODS):ax.text(i,ci[i,1]+3,f"{s['methods'][m]['successes']}/50",ha='center',fontsize=10)
        fig.text(.13,.015,'50 paired throws · 95% Wilson intervals · Frozen simulation model',fontsize=8,color='#555555')
        fig.tight_layout(rect=(0,.06,1,1))
        for extension in ['svg','png']:fig.savefig(args.output/f'{route}-success-rates.{extension}',dpi=200)
        plt.close(fig)
        fig,ax=plt.subplots(figsize=(5.2,4.3))
        selected=[r for r in rows if r['route']==route]
        for i,m in enumerate(METHODS):
            x=np.sort([float(r['clean_hold_s'])for r in selected if r['method']==m])
            ax.step(np.r_[0,x],np.r_[0,np.arange(1,51)/50]*100,where='post',label=LABELS[route][i].replace('\n',' '),color=COLORS[i],linestyle=['-','--',':'][i],linewidth=2)
        limit=max(.05,1.1*max(float(r['clean_hold_s'])for r in selected))
        ax.set(xlim=(0,limit),ylim=(0,103),xlabel='Longest uninterrupted clean bilateral control (s)',ylabel='Empirical cumulative probability (%)',title=route.capitalize())
        ax.legend(fontsize=8,frameon=False,loc='lower right');ax.grid(alpha=.15)
        note='Route specific time scale; 0.75 s criterion lies outside view.'if limit<.75 else'Control duration alone does not enforce terminal possession.'
        fig.text(.13,.015,note,fontsize=8,color='#555555');fig.tight_layout(rect=(0,.06,1,1))
        for extension in ['svg','png']:fig.savefig(args.output/f'{route}-control-duration-ecdf.{extension}',dpi=200)
        plt.close(fig)
    print(json.dumps({'rates':exported,'exploratory_paired_contrasts':contrasts},indent=2))


if __name__=='__main__':main()
