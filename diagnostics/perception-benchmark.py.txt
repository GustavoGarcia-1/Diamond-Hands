"""Paired image benchmark. Replay state is rendering/evaluation only.

No simulator integration, control, reward, training or physical outcome occurs
here. Both observers receive the same immutable RGB-D and surveyed pose packet.
"""
from pathlib import Path
from dataclasses import replace
import csv, hashlib, importlib.util, json, os, sys
import xml.etree.ElementTree as ET
os.environ['MUJOCO_GL']='egl'
import numpy as np
import mujoco
from PIL import Image

base,output=map(Path,sys.argv[1:3]);output.mkdir(parents=True,exist_ok=False)
sys.path[:0]=[str(base/'source/scripts'),str(base/'source/src')]
from g1_receiver.perception import CameraIntrinsics,RGBDFrame,RigidPose,ColorBallDetector
from g1_receiver.sim import WIDTH,HEIGHT
from evaluate_aero_receiver import fixed_camera
from evaluate_dextra_in_stride import RearMarkerLocalizer
spec=importlib.util.spec_from_file_location('g1_receiver.associated_markers',Path(__file__).with_name('associated_markers.py'))
association=importlib.util.module_from_spec(spec);sys.modules[spec.name]=association;spec.loader.exec_module(association)
protocol=json.loads(Path(__file__).with_name('protocol.json').read_text())
assert protocol['frozen_before_execution']
records=[];manifests=[];inputs=[]
times=np.r_[np.arange(4.4,10.91,.1),10.98]
conditions=protocol['conditions']
for trace_index,policy in enumerate(['zero','learned']):
    episode=base/'artifacts/ppo/evaluation/nominal'/policy/'episode-0001'
    scene_path=episode/'trial/scene.xml';trace_path=episode/'trial/actual-state.npz'
    inputs.append(dict(policy=policy,scene_sha256=hashlib.sha256(scene_path.read_bytes()).hexdigest(),
                       trace_sha256=hashlib.sha256(trace_path.read_bytes()).hexdigest()))
    scene=ET.parse(scene_path).getroot()
    for name,coords in protocol['extra_views'].items():fixed_camera(scene,name,*coords)
    model=mujoco.MjModel.from_xml_string(ET.tostring(scene,encoding='unicode'))
    data=mujoco.MjData(model);renderer=mujoco.Renderer(model,height=HEIGHT,width=WIDTH)
    raw=np.load(trace_path)['time_qpos'];original_rgba=model.geom_rgba.copy()
    for visibility_index,visibility in enumerate(['intact','yellow_marker_hidden','ball_hidden']):
        model.geom_rgba[:]=original_rgba
        if visibility=='yellow_marker_hidden':model.geom_rgba[model.geom('rear_marker_yellow').id,3]=0.
        if visibility=='ball_hidden':model.geom_rgba[model.geom('football_surface').id,3]=0.
        for frame_index,t in enumerate(times):
            state=raw[np.argmin(abs(raw[:,0]-t))];data.qpos[:]=state[1:];data.time=state[0]
            mujoco.mj_forward(model,data)
            # Evaluation label saved separately; neither observer receives it.
            truth=data.body('pelvis').xpos.copy();rotation=data.body('pelvis').xmat.reshape(3,3)
            true_yaw=float(np.arctan2(rotation[1,0],rotation[0,0]))
            for camera_index,camera in enumerate(protocol['views']):
                cid=model.camera(camera).id
                renderer.update_scene(data,camera=camera);rgb=renderer.render()
                renderer.enable_depth_rendering();depth=renderer.render();renderer.disable_depth_rendering()
                focal=.5*HEIGHT/np.tan(np.deg2rad(model.cam_fovy[cid])*.5)
                intrinsics=CameraIntrinsics(focal,focal,(WIDTH-1)/2,(HEIGHT-1)/2)
                survey=RigidPose(float(state[0]),'camera_optical','field',data.cam_xmat[cid].reshape(3,3)@np.diag([1.,-1.,-1.]),data.cam_xpos[cid].copy(),'synthetic_fixture')
                base_frame=RGBDFrame(float(state[0]),'camera_optical',rgb,depth,intrinsics)
                manifests.append(dict(policy=policy,visibility=visibility,time_s=float(state[0]),camera=camera,
                    rgb_sha256=hashlib.sha256(rgb.tobytes()).hexdigest(),depth_sha256=hashlib.sha256(depth.tobytes()).hexdigest()))
                if frame_index in [0,len(times)-1] and camera in ['retention','late_pose']:
                    Image.fromarray(rgb).save(output/f'{policy}-{visibility}-{frame_index:03d}-{camera}.png')
                for condition_index,condition in enumerate(conditions):
                    name=condition['name'];image=rgb.copy();z=depth.copy();pose=survey;now=float(state[0]);timestamp=now
                    rng=np.random.default_rng(np.random.SeedSequence([protocol['seed'],trace_index,visibility_index,frame_index,camera_index,condition_index]))
                    if 'brightness' in condition:image=np.clip(np.rint(image.astype(float)*condition['brightness']),0,255).astype(np.uint8)
                    if 'depth_bias_m' in condition:z=z+condition['depth_bias_m']
                    if 'depth_noise_std_m' in condition:z=z+rng.normal(0,condition['depth_noise_std_m'],z.shape)
                    if 'depth_dropout_fraction' in condition:z[rng.random(z.shape)<condition['depth_dropout_fraction']]=np.nan
                    if 'survey_x_bias_m' in condition:pose=replace(pose,translation=pose.translation+np.array([condition['survey_x_bias_m'],0.,0.]))
                    if 'latency_s' in condition:now+=condition['latency_s']
                    if 'future_timestamp_s' in condition:timestamp+=condition['future_timestamp_s'];pose=replace(pose,timestamp_s=timestamp)
                    frame=replace(base_frame,rgb=image,depth_m=z,timestamp_s=timestamp)
                    frame.rgb.flags.writeable=False;frame.depth_m.flags.writeable=False
                    packet_hash=hashlib.sha256(image.tobytes()+z.tobytes()+pose.rotation.tobytes()+pose.translation.tobytes()+np.array([now,timestamp]).tobytes()).hexdigest()
                    ball=ColorBallDetector(max_depth_m=20.,ball_center_offset_m=.075).detect(frame,pose)
                    for method,cls in [('original',RearMarkerLocalizer),('rig_association',association.AssociatedRearMarkerLocalizer)]:
                        observer=cls(allow_synthetic_fixture=True,max_range_m=8.)
                        estimate=observer.estimate(frame,pose)
                        usable=estimate is not None and estimate.fresh(now,.08000001)
                        error=None if estimate is None else float(np.linalg.norm(np.r_[estimate.xy_m,estimate.z_m]-truth))
                        yaw_error=None if estimate is None else float(abs(np.arctan2(np.sin(estimate.yaw_rad-true_yaw),np.cos(estimate.yaw_rad-true_yaw))))
                        records.append(dict(policy=policy,visibility=visibility,condition=name,camera=camera,
                            requested_time_s=float(t),capture_s=float(timestamp),now_s=now,method=method,
                            packet_sha256=packet_hash,pose_returned=estimate is not None,pose_usable=usable,
                            xyz_error_m=error,yaw_error_rad=yaw_error,ball_detected=ball is not None,
                            missing_marker_false_pose=visibility=='yellow_marker_hidden' and usable,
                            absent_ball_false_detection=visibility=='ball_hidden' and ball is not None,
                            gross_pose_error=usable and error>.10))
                assert data.time==state[0] and np.array_equal(data.qpos,state[1:])
        print(json.dumps({'completed_policy':policy,'visibility':visibility,'rows':len(records)}),flush=True)
    renderer.close()

with(output/'paired-observations.csv').open('w')as stream:
    writer=csv.DictWriter(stream,fieldnames=list(records[0]));writer.writeheader();writer.writerows(records)
groups={}
for row in records:
    key=(row['policy'],row['visibility'],row['condition'],row['camera'],row['method'])
    groups.setdefault(key,[]).append(row)
summary=[]
for key,rows in groups.items():
    errors=[r['xyz_error_m']for r in rows if r['pose_usable']]
    summary.append(dict(zip(['policy','visibility','condition','camera','method'],key),
        frames=len(rows),pose_usable=sum(r['pose_usable']for r in rows),
        missing_marker_false_pose=sum(r['missing_marker_false_pose']for r in rows),
        absent_ball_false_detection=sum(r['absent_ball_false_detection']for r in rows),
        gross_pose_error=sum(r['gross_pose_error']for r in rows),
        xyz_rmse_m=None if not errors else float(np.sqrt(np.mean(np.square(errors)))),
        xyz_p95_m=None if not errors else float(np.quantile(errors,.95))))
for name,obj in [('summary.json',summary),('frame-manifest.json',manifests),('input-manifest.json',inputs)]:
    (output/name).write_text(json.dumps(obj,indent=2)+'\n')
assert all(a['packet_sha256']==b['packet_sha256']for a,b in zip(records[::2],records[1::2]))
(output/'receipt.json').write_text(json.dumps(dict(schema='paired-marker-perception-benchmark-v1',rows=len(records),
    all_observer_pairs_identical_input=True,no_simulation_steps=True,no_control_or_reward=True,
    script_sha256=hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
    observer_sha256=hashlib.sha256(Path(association.__file__).read_bytes()).hexdigest(),
    protocol_sha256=hashlib.sha256(Path(__file__).with_name('protocol.json').read_bytes()).hexdigest(),
    uncertainty_scope='Descriptive paired frame statistics from two deterministic Go trajectories. Frames are correlated; no iid-frame confidence interval, catch success rate or hardware generalization claim.'),indent=2)+'\n')
